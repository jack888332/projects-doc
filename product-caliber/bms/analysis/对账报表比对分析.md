# 数据分析

## 财务侧报表与系统侧报表比对 SQL

说明：该 SQL 用于生成财务侧报表与系统侧报表的订单级费项比对结果。财务侧临时表沿用现有财务报表格式；系统侧临时表必须先从新版应收账单对账报表的`费项明细`中识别业务订单父行，移除业务订单号中的层级标志，并排除尾程包裹子行和结构异常订单。实际比对时，费项比对列并非固定清单，而是财务侧报表和系统侧报表的费项并集；每个费项按 `<费项><财务侧>`、`<费项><系统侧>` 成对输出。

```sql
WITH 
-- 财务侧报表和系统侧报表的业务订单并集
U1 AS (
  SELECT
    IFNULL(T1.`订单号`, T2.`业务订单号`) AS `业务订单号`,
    IFNULL(DATE_FORMAT(T1.`出库时间`, '%Y/%m/%d'), DATE_FORMAT(T2.`出库时间`, '%Y/%m/%d')) AS `出库时间`,
    IF(T1.`订单号` IS NOT NULL, '✅', NULL) AS `财务侧报表`,
    IF(T2.`业务订单号` IS NOT NULL, '✅', NULL) AS `系统侧报表`
  FROM
    myspace.`财务侧报表` AS T1
    LEFT JOIN myspace.`系统侧报表` AS T2 ON T1.`订单号` = T2.`业务订单号` 
    WHERE T2.`业务订单号` REGEXP '.+(?<!\>)$'
  UNION
  SELECT
    IFNULL(T1.`订单号`, T2.`业务订单号`) AS `订单号`,
    IFNULL(DATE_FORMAT(T1.`出库时间`, '%Y/%m/%d'), DATE_FORMAT(T2.`出库时间`, '%Y/%m/%d')) AS `出库时间`,
    IF(T1.`订单号` IS NOT NULL, '✅', NULL) AS `财务侧报表`,
    IF(T2.`业务订单号` IS NOT NULL, '✅', NULL) AS `系统侧报表`
  FROM
    myspace.`财务侧报表` AS T1
    RIGHT JOIN myspace.`系统侧报表` AS T2 ON T1.`订单号` = T2.`业务订单号`
    WHERE T2.`业务订单号` REGEXP '.+(?<!\>)$'
),
-- 运费统计
U2 AS (
  SELECT
    U1.`业务订单号`,
    ROUND(SUM(T1.`运费`), 3) AS `运费`
  FROM
    U1
    LEFT JOIN myspace.`财务侧报表` AS T1 ON T1.`订单号` = U1.`业务订单号`
  GROUP BY
    U1.`业务订单号`
),
-- 派送费统计
U21 AS (
  SELECT
    U1.`业务订单号`,
    ROUND(SUM(T1.`派送费`), 3) AS `派送费`
  FROM
    U1
    LEFT JOIN myspace.`财务侧报表` AS T1 ON T1.`订单号` = U1.`业务订单号`
  GROUP BY
    U1.`业务订单号`
),
-- 超材费统计
U3 AS (
  SELECT
    U1.`业务订单号`,
    ROUND(SUM(T1.`其它费用`), 3) AS `超材费`
  FROM
    U1
    LEFT JOIN myspace.`财务侧报表` AS T1 ON
    IF(T1.`订单号` != '/', T1.`订单号`, T1.`承运单号`) = U1.`业务订单号`
  WHERE
    T1.`收件人地址` = '超材费'
  GROUP BY
    U1.`业务订单号`
),
-- 转板费统计
U4 AS (
  SELECT
    U1.`业务订单号`,
    ROUND(SUM(T1.`其它费用`), 3) AS `转板费`
  FROM
    U1
    LEFT JOIN myspace.`财务侧报表` AS T1 ON
    IF(T1.`订单号` != '/', T1.`订单号`, T1.`承运单号`) = U1.`业务订单号`
  WHERE
    T1.`收件人地址` = '转板费'
  GROUP BY
    U1.`业务订单号`
),
-- 税金统计
U5 AS (
  SELECT
    U1.`业务订单号`,
    ROUND(SUM(T1.`其它费用`), 3) AS `税金`
  FROM
    U1
    LEFT JOIN myspace.`财务侧报表` AS T1 ON
    IF(T1.`订单号` != '/', T1.`订单号`, T1.`承运单号`) = U1.`业务订单号`
  WHERE
    T1.`收件人地址` = '税金'
  GROUP BY
    U1.`业务订单号`
),
-- 代收货款统计
U6 AS (
  SELECT
    U1.`业务订单号`,
    ROUND(SUM(T1.`应收货款`), 3) AS `代收货款`,
    ROUND(SUM(T1.`手续费`), 3) AS `货到付款手续费`
  FROM
    U1
    LEFT JOIN myspace.`财务侧报表` AS T1 ON
    IF(T1.`订单号` != '/', T1.`订单号`, T1.`承运单号`) = U1.`业务订单号`
  WHERE
    T1.`收件人地址` = '代收返款'
  GROUP BY
    U1.`业务订单号`
) 
-- 组装比对结果
 SELECT
  U1.*,
  U2.`运费` AS `运费<财务侧>`,
  T1.`运费` AS `运费<系统侧>`,
  U21.`派送费` AS `派送费<财务侧>`,
  T1.`派送费` AS `派送费<系统侧>`,  
  U3.`超材费` AS `超材费<财务侧>`,
  T1.`超材费` AS `超材费<系统侧>`,
  U4.`转板费` AS `转板费<财务侧>`,
  T1.`转板费` AS `转板费<系统侧>`,
  U5.`税金` AS `税金<财务侧>`,
  T1.`税金` AS `税金<系统侧>`,
  U6.`代收货款` AS `代收货款<财务侧>`,
  T1.`代收货款` AS `代收货款<系统侧>`,
  U6.`货到付款手续费` AS `货到付款手续费<财务侧>`,
  T1.`货到付款手续费` AS `货到付款手续费<系统侧>`
FROM
  U1
  LEFT JOIN myspace.`系统侧报表` AS T1 ON T1.`业务订单号` = U1.`业务订单号`
  LEFT JOIN U2 ON U2.`订单号` = U1.`业务订单号`
  LEFT JOIN U21 ON U21.`订单号` = U1.`业务订单号`
  LEFT JOIN U3 ON U3.`订单号` = U1.`业务订单号`
  LEFT JOIN U4 ON U4.`订单号` = U1.`业务订单号`
  LEFT JOIN U5 ON U5.`订单号` = U1.`业务订单号`
  LEFT JOIN U6 ON U6.`订单号` = U1.`业务订单号`
WHERE
  U1.`业务订单号` != '/'
ORDER BY
  U1.`出库时间` DESC,
  U1.`业务订单号` DESC;
```
